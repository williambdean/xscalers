preprocessing

Transformers, mirroring sklearn.preprocessing.

Two families live here, both of which sklearn also keeps in sklearn.preprocessing:

  • the scalers — ZScore, MeanScaler, FixedScaler (all affine, y = (x - center) / scale) and Func for named elementwise ops (log, log1p, ... — the FunctionTransformer analog);
  • the steps that build, split or select variables rather than rescaling them — Derive (a new variable from selected pieces, the PolynomialFeatures analog), Splitter and Selector.

Everything affine exposes state keys center and scale (z-score's mean/std map onto this pair, matching PR-2831).

Derive

Derive a new variable from selected pieces of existing variables.

Pieces reference (var_name, sel), so the two sides need not come from the same variable or the same grain — xarray broadcasting does the rest. A national series over a product-grain control, giving a product-grain derived variable:

Derive(
    name="tv_per_control",
    left=("media_national", {"media": "tv"}),
    right=("controls_prod", {}),
    op="ratio",
)

The within-one-variable shape works the same way, selecting coordinate values out of a variable that carries an extra coordinate dimension:

Derive(
    name="level_ratio",
    left=("controls", {"control": "level"}),
    right=("controls", {"control": "reference"}),
    op="ratio",
)
  • selectors are the same Splitter-style label dict, so lists + reduce (sum | mean | first | last) cover share-of-total style derivations, while scalar selections pass through directly.
  • ops: ratio | diff | product — a closed named vocabulary, because specs must stay plain JSON (no lambdas).
  • stateless: fit records only the recipe; inverse_transform drops the derived variable (its inputs survive untouched), so a pipeline containing it still round-trips.
  • Dataset inputs only (DataTree derivation is deferred; the derived var needs a flat variable namespace).

class Derive:
    """Derive a new variable from selected pieces of existing variables.

    Pieces reference ``(var_name, sel)``, so the two sides need not come from
    the same variable or the same grain — xarray broadcasting does the rest.
    A national series over a product-grain control, giving a product-grain
    derived variable:

        Derive(
            name="tv_per_control",
            left=("media_national", {"media": "tv"}),
            right=("controls_prod", {}),
            op="ratio",
        )

    The within-one-variable shape works the same way, selecting coordinate
    values out of a variable that carries an extra coordinate dimension:

        Derive(
            name="level_ratio",
            left=("controls", {"control": "level"}),
            right=("controls", {"control": "reference"}),
            op="ratio",
        )

    - selectors are the same ``Splitter``-style label dict, so lists +
      ``reduce`` (sum | mean | first | last) cover share-of-total style
      derivations, while scalar selections pass through directly.
    - ops: ``ratio`` | ``diff`` | ``product`` — a closed named vocabulary,
      because specs must stay plain JSON (no lambdas).
    - stateless: ``fit`` records only the recipe; ``inverse_transform``
      drops the derived variable (its inputs survive untouched), so a
      pipeline containing it still round-trips.
    - Dataset inputs only (DataTree derivation is deferred; the derived var
      needs a flat variable namespace).
    """

    def __init__(
        self,
        name: str,
        left: tuple[str, Mapping[str, Any]] | None = None,
        right: tuple[str, Mapping[str, Any]] | None = None,
        *,
        op: str = "ratio",
        reduce: str | None = None,
    ):
        """``left op right`` — neutral names over the closed op vocabulary."""
        if op not in {"ratio", "diff", "product"}:
            raise ValueError(f"Derive ops are ratio | diff | product; got {op!r}")
        if right is None:
            raise ValueError(f"Derive op={op!r} needs a right piece")
        if reduce is not None and reduce not in {"sum", "mean", "first", "last"}:
            raise ValueError(f"unsupported reduce {reduce!r}; use sum | mean | first | last")
        self.name: str = name
        self.left = self._normalize_piece(left)
        self.right = self._normalize_piece(right)
        self.op = op
        self.reduce = reduce

    @staticmethod
    def _normalize_piece(piece: Any) -> tuple[str, dict[str, Any]]:
        """Accept ``(var, sel)`` tuples or serialized ``{"var", "sel"}`` dicts."""
        if isinstance(piece, dict):
            return str(piece["var"]), dict(piece["sel"])
        return str(piece[0]), dict(piece[1])

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize the derive recipe into JSON-safe kwargs."""
        return {
            "name": self.name,
            "left": {"var": self.left[0], "sel": self._jsonable_sel(self.left[1])},
            "right": None
            if self.right is None
            else {"var": self.right[0], "sel": self._jsonable_sel(self.right[1])},
            "op": self.op,
            "reduce": self.reduce,
        }

    @staticmethod
    def _jsonable_sel(sel: Mapping[str, Any]) -> dict[str, Any]:
        return {dim: list(value) if isinstance(value, (list, tuple)) else value for dim, value in sel.items()}

    def to_spec(self) -> dict[str, Any]:
        """Serialize the derive recipe."""
        from .serialize import spec_of

        return spec_of(self)

    # -- pieces --------------------------------------------------------------

    def _piece(self, X: xr.Dataset, piece: tuple[str, Mapping[str, Any]] | None) -> xr.DataArray:
        var, sel = piece if piece is not None else (None, {})
        da = X[var]
        sel_args: dict[str, Any] = {
            str(dim): list(value) if isinstance(value, (list, tuple)) else value for dim, value in sel.items()
        }
        for dim, value in sel_args.items():
            da = da.sel(**{dim: value})
        reduce = self.reduce
        needs_reduce = any(isinstance(v, (list, tuple)) and len(v) > 1 for v in sel.values())
        if reduce and needs_reduce:
            da = getattr(da, reduce)(dim=list(sel))
        elif reduce is not None and needs_reduce:
            raise ValueError(f"reduce={reduce!r} is not a recognized reducer")
        return da

    # -- transformer API -------------------------------------------------------

    def fit(self, X: Any) -> xr.Dataset:
        """Validate pieces exist, then record the derived name and op."""
        if not isinstance(X, xr.Dataset):
            raise TypeError("Derive works on Dataset inputs (DataTree derivation deferred)")
        for piece in (self.left, self.right):
            if piece is None:
                continue
            var = piece[0]
            if var not in X.data_vars:
                raise KeyError(f"Derive piece references unknown variable {var!r}")
        derived = self.name
        if X is not None and derived in X.data_vars:
            raise ValueError(f"Derived variable name {derived!r} already exists")
        return _stamp_state(
            as_state(
                "derive",
                None,
                scaled_from=None,
                attrs={
                    "derived": self.name,
                    "op": self.op,
                    "reduce": self.reduce or "",
                },
            ),
            self,
        )

    @classmethod
    def describe(cls, spec):
        """One created row for the derived variable, affine by op."""
        from .serialize import _recipe_line

        return [
            {
                "variable": spec["name"],
                "role": "created",
                "kind": "Derive",
                "recipe": _recipe_line({"kind": "Derive", "kwargs": spec}),
                "op": str(spec.get("op", "") or ""),
                "affine": True,
            }
        ]

    def transform(self, X: Any, state: xr.Dataset) -> Any:
        """Select the two pieces, apply the op, and append the new variable."""
        if not isinstance(X, xr.Dataset):
            raise TypeError("Derive works on Dataset inputs")
        first = self._piece(X, self.left)
        second = self._piece(X, self.right)
        if self.op == "ratio":
            derived = first / second
        elif self.op == "diff":
            derived = first - second
        else:
            derived = first * second
        out = {key: da for key, da in flatten(X).items()}
        out[self.name] = derived.rename(self.name)
        return reassemble(X, out)

    def inverse_transform(self, X: Any, state: xr.Dataset) -> Any:
        """Drop the derived variable, restoring the original variables."""
        out = {key: da for key, da in flatten(X).items() if key != state.attrs["derived"]}
        return reassemble(X, out)

Causal exponential smoothing: s_t = a*x_t + (1-a)*s_{t-1}.

Seeds at the first value (s_0 = x_0), so smoothing a fresh block of data burns in from that block's own values and never yields NaN, mirroring RollingMean. Out-of-sample: transform needs only the fitted alpha (state) and the X it is handed; for boundary continuity concat the in-sample tail with the new block and slice afterwards.

One named dim to smooth over: dims=("date",) or equivalent; an ambiguous dims=None on multi-dim data raises. Internal NaNs in X propagate through the recursion (a missing value never recovers).

Not affine: Pipeline.affine() refuses pipelines containing it, and inverse_transform honestly raises. xarray-only: no symbolic (pytensor.xtensor) path.

class EwmSmooth(BaseTransformer):
    """Causal exponential smoothing: ``s_t = a*x_t + (1-a)*s_{t-1}``.

    Seeds at the first value (``s_0 = x_0``), so smoothing a fresh block of
    data burns in from that block's own values and never yields NaN, mirroring
    ``RollingMean``. Out-of-sample: ``transform`` needs only the fitted
    ``alpha`` (state) and the X it is handed; for boundary continuity concat
    the in-sample tail with the new block and slice afterwards.

    One named dim to smooth over: ``dims=("date",)`` or equivalent; an
    ambiguous ``dims=None`` on multi-dim data raises. Internal NaNs in X
    propagate through the recursion (a missing value never recovers).

    Not affine: ``Pipeline.affine()`` refuses pipelines containing it, and
    ``inverse_transform`` honestly raises. xarray-only: no symbolic
    (``pytensor.xtensor``) path.
    """

    def __init__(self, alpha: float = 0.5, dims: tuple[str, ...] | str | None = None):
        """Take the recursion weight ``alpha`` in ``(0, 1]`` and the dim."""
        if not 0 < alpha <= 1:
            raise ValueError(f"alpha must be in (0, 1]; got {alpha}")
        super().__init__(dims)
        self.alpha = alpha

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize alpha and dims into plain-dict kwargs."""
        return {"alpha": self.alpha, "dims": self.dims}

    def compute(self, X: Any) -> xr.Dataset:
        """Record the recursion weight (the only fitted knowledge)."""
        if is_symbolic(X):
            raise NotImplementedError("EwmSmooth is xarray-only (no symbolic path)")
        self._smooth_dim(X)
        scaled_from = getattr(X, "name", None)
        return as_state("ewm", self.dims, scaled_from, attrs={"alpha": self.alpha})

    def _smooth_dim(self, X: Any) -> str:
        """Resolve the single dim the recursion runs along."""
        dims = self._reduce_dims(X)
        if len(dims) != 1:
            raise ValueError(
                f"EwmSmooth smooths exactly one dim; resolved {dims}. Pass dims=<dim name> explicitly."
            )
        return dims[0]

    def _transform(self, X: Any, state: Mapping[str, xr.DataArray] | xr.Dataset) -> Any:
        if is_symbolic(X):
            raise NotImplementedError("EwmSmooth is xarray-only (no symbolic path)")
        alpha = float(getattr(state, "attrs", {}).get("alpha", self.alpha))
        axis = X.get_axis_num(self._smooth_dim(X))
        values = np.moveaxis(np.asarray(X, dtype=float), axis, 0)
        decay = 1.0 - alpha
        out = np.empty(values.shape, dtype=float)
        out[0] = values[0]
        prev = out[0]
        for t in range(1, values.shape[0]):
            prev = alpha * values[t] + decay * prev
            out[t] = prev
        out = np.moveaxis(out, 0, axis)
        return xr.DataArray(out, coords=X.coords, dims=X.dims).assign_attrs(X.attrs)

    def _inverse_transform(self, X: Any, state: Mapping[str, xr.DataArray] | xr.Dataset) -> Any:
        """Smoothing mixes history; raising is the contract, not a guess."""
        raise NonInvertibleError(
            "EwmSmooth recursively mixes history; inverse_transform cannot "
            "honestly restore the original series (the seed is lost)."
        )

class FixedScaler(BaseAffineTransformer):
    """Constant affine pair — scale by a fixed amount (e.g. scale by 100).

    No data-derived stats: ``compute`` returns constants, so the same object
    also serves as a consumer's fixed-scale contract.
    """

    def __init__(self, scale: float = 100.0, center: float = 0.0):
        """Pin the affine pair; no data is consulted."""
        super().__init__(dims=None)
        self.scale = scale
        self.center = center

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize the constants into plain-dict kwargs."""
        return {"scale": self.scale, "center": self.center}

    def compute(self, X: Any) -> xr.Dataset:
        """Return the constant pair, stamped with what it scales."""

class Loess(BaseTransformer):
    """Local regression over a trailing or centered window.

    Default ``center=False``: each position's value is a weighted
    least-squares fit of the trailing ``window`` points evaluated at the
    current point, causal and out-of-sample safe exactly like
    ``RollingMean``: lead-in windows shrink and are renormalized, no NaN.
    ``center=True`` fits over a symmetric window evaluated at its middle,
    the textbook (two-sided) loess look: far smoother, but near a block's
    right edge the window degenerates to one-sided (no future data).
    ``degree=1`` is the loess-like local linear fit, ``degree=0`` is the
    kernel-weighted mean.

    One named dim to regress along; ambiguous ``dims=None`` on multi-dim
    data raises. Missing values inside a window are excluded (weights
    renormalized); a window with no valid points falls back to pure xarray
    arithmetic honestly (NaN out).

    Not affine: ``Pipeline.affine()`` refuses pipelines containing it, and
    ``inverse_transform`` honestly raises. xarray-only: no symbolic path.
    """

    def __init__(
        self,
        window: int = 4,
        degree: int = 1,
        dims: tuple[str, ...] | str | None = None,
        center: bool = False,
    ):
        """Take the window length, fit degree (0 | 1), the dim, and centering."""
        if window < 1:
            raise ValueError(f"window must be >= 1; got {window}")
        if degree not in (0, 1):
            raise ValueError(f"degree must be 0 or 1; got {degree}")
        super().__init__(dims)
        self.window = window
        self.degree = degree
        self.center = center

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize window, degree, centering and dims into plain kwargs."""
        return {
            "window": self.window,
            "degree": self.degree,
            "dims": self.dims,
            "center": self.center,
        }

    def compute(self, X: Any) -> xr.Dataset:
        """Record window and degree (the only fitted knowledge)."""
        if is_symbolic(X):
            raise NotImplementedError("Loess is xarray-only (no symbolic path)")
        self._smooth_dim(X)
        scaled_from = getattr(X, "name", None)
        return as_state(
            "loess",
            self.dims,
            scaled_from,
            attrs={
                "window": self.window,
                "degree": self.degree,
                "center": int(self.center),
            },
        )

    def _smooth_dim(self, X: Any) -> str:
        """Resolve the single dim the regression runs along."""
        dims = self._reduce_dims(X)
        if len(dims) != 1:
            raise ValueError(
                f"Loess smooths exactly one dim; resolved {dims}. Pass dims=<dim name> explicitly."
            )
        return dims[0]

    @staticmethod
    def _tricube(distances: xr.DataArray, max_distance: float) -> xr.DataArray:
        """Tricube kernel weights over lag distances (0 is the anchor)."""
        if max_distance > 0:
            return (1 - (distances / max_distance) ** 3) ** 3
        return xr.ones_like(distances)

    def _transform(self, X: Any, state: Mapping[str, xr.DataArray] | xr.Dataset) -> Any:
        if is_symbolic(X):
            raise NotImplementedError("Loess is xarray-only (no symbolic path)")
        attrs = getattr(state, "attrs", {})
        window = int(attrs.get("window", self.window))
        degree = int(attrs.get("degree", self.degree))
        center = int(attrs.get("center", int(self.center))) == 1
        dim = self._smooth_dim(X)
        attrs_in = dict(X.attrs)
        windows = X.rolling({dim: window}, center=center).construct({dim: "lag"})
        # lag index of the evaluation anchor: the newest point (trailing) or
        # the window's middle (centered); |offset| drives the kernel, the
        # signed offset drives the degree-1 design (a line is exact in it)
        anchor = window - 1 if not center else window // 2
        offsets = xr.DataArray(np.arange(window, dtype=float) - anchor, dims="lag")
        distances = xr.DataArray(np.abs(offsets.values), dims="lag")
        weights = self._tricube(distances, max_distance=float(window - 1))
        valid = xr.DataArray(np.isfinite(windows.values), dims=windows.dims)
        weights = weights * valid  # drop missing positions from every sum
        total = weights.sum("lag")
        numerator = (windows * weights).sum("lag")
        if degree == 0:
            return (numerator / total).assign_attrs(attrs_in)
        # local linear fit evaluated at offset 0: a and b solve the
        # weighted normal equations in closed form over the lag dim
        weighted = weights * offsets
        s0 = total
        s1 = weighted.sum("lag")
        s2 = (weighted * offsets).sum("lag")
        sy = numerator
        szy = (windows * weighted).sum("lag")
        determinant = s0 * s2 - s1 * s1
        intercept = (sy * s2 - szy * s1) / determinant
        return xr.where(determinant > 0, intercept, numerator / total)

    def _inverse_transform(self, X: Any, state: Mapping[str, xr.DataArray] | xr.Dataset) -> Any:
        """Overlapping local fits overlap information; raising is the contract."""
        raise NonInvertibleError(
            "Loess fits overlapping windows; inverse_transform cannot "
            "honestly restore the original series (the lead-in is lost)."
        )

class Splitter:
    """Split one variable along a coordinate into named piece nodes.

    ``pieces`` maps piece name → sel-dict along one dim, e.g.

        Splitter("media", {"digital": {"channel": ["search", "social"]},
                           "tv": {"channel": ["tv"]}})

    The original variable is replaced by child nodes ``media/digital``,
    ``media/tv`` (var renamed to the piece name), so a following
    ``VariableTransformer`` step scales each piece independently. DataTree requires
    child coords to not conflict with the parent's, so ``transform`` removes
    the source var from the source node's dataset.
    """

    def __init__(self, source: str, pieces: Mapping[str, Mapping[str, Any]]):
        """Take the source variable and ``{piece: {dim: labels}}`` selectors.

        All pieces must split along exactly one shared dim.
        """
        self.source = source
        # a bare label string means one label, the way Selector/Derive accept
        # ``{"media": "tv"}`` — otherwise a str would be iterated per character
        self.pieces = {piece: dict(self._labels(sel)) for piece, sel in pieces.items()}
        split_dims = {dim for sel in self.pieces.values() for dim in sel}
        if len(split_dims) != 1:
            raise ValueError(f"Splitter splits along exactly one dim; pieces give {sorted(split_dims)}")
        self.split_dim = next(iter(split_dims))

    @staticmethod
    def _labels(sel: Mapping[str, Any]) -> dict[str, list[Any]]:
        return {dim: [value] if isinstance(value, str) else list(value) for dim, value in sel.items()}

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize the split recipe into plain-dict kwargs."""
        return {
            "source": self.source,
            "pieces": {piece: dict(sel) for piece, sel in self.pieces.items()},
        }

    # -- transformer API ---------------------------------------------------

    def fit(self, X: xr.DataTree | xr.Dataset) -> xr.Dataset:
        """Store the coordinate labels and per-piece positions.

        The state also records the exact output variable names (``piece_vars``)
        so the affine view matches the container's naming convention.
        """
        var = self.source.split("/")[-1]
        if isinstance(X, xr.Dataset):
            labels = np.asarray(X[var][self.split_dim].values)
            # Dataset outputs: sibling variables {var}_{piece} (see transform)
            piece_vars = [f"{var}_{piece}" for piece in self.pieces]
        elif isinstance(X, xr.DataTree):
            node = _node_of(X, self.source)
            labels = np.asarray(node[var][self.split_dim].values)
            # DataTree outputs: piece nodes under the source node; keys are
            # node paths, matching how transform/flatten name them
            piece_vars = [f"{self.source}/{piece}" for piece in self.pieces]
        else:
            raise TypeError("Splitter works on Dataset or DataTree inputs")
        entries: dict[str, Any] = {f"labels_{self.split_dim}": xr.DataArray(labels, dims=("labels_index",))}
        for piece, sel in self.pieces.items():
            entries[f"positions_{piece}"] = xr.DataArray(
                _positions(labels, sel[self.split_dim]), dims=(f"index_{piece}",)
            )
        return _stamp_state(
            as_state(
                "splitter",
                None,
                scaled_from=self.source,
                attrs={
                    "source": self.source,
                    "split_dim": self.split_dim,
                    "pieces": sorted(self.pieces),
                    # the exact output variable names this fit produces, so
                    # affine_view (state-side) matches the container's naming
                    "piece_vars": list(piece_vars),
                },
                **entries,
            ),
            self,
        )

    def affine_view(self, state: xr.Dataset) -> dict[str, xr.Dataset]:
        """Identity pairs for the created pieces.

        The split is a selection, value-preserving, so the composed affine
        answer composes through it. The source variable is gone post-step, so
        it gets no pair.
        """
        piece_vars = state.attrs.get("piece_vars")
        if piece_vars is None:  # older state: Dataset naming convention
            var = str(state.attrs["source"]).split("/")[-1]
            piece_vars = [f"{var}_{piece}" for piece in state.attrs["pieces"]]
        return {
            name: xr.Dataset({"center": xr.DataArray(0.0), "scale": xr.DataArray(1.0)}) for name in piece_vars
        }

    @classmethod
    def describe(cls, spec):
        """One split row for the source, one created row per piece."""
        from .serialize import _recipe_line

        recipe = _recipe_line({"kind": "Splitter", "kwargs": spec})
        rows = [
            {
                "variable": str(spec["source"]).split("/")[-1],
                "role": "split",
                "kind": "Splitter",
                "recipe": recipe,
                "op": "",
                "affine": True,
            }
        ]
        var = str(spec["source"]).split("/")[-1]
        rows += [
            {
                "variable": f"{var}_{piece}",
                "role": "created",
                "kind": "Splitter",
                "recipe": recipe,
                "op": "",
                "affine": True,
            }
            for piece in spec["pieces"]
        ]
        return rows

    def transform(self, X: xr.DataTree | xr.Dataset, state: xr.Dataset):
        """Replace the source variable with its named pieces.

        Dataset inputs rename each piece's split dim (``{dim}_{piece}``) so
        sibling variables can coexist; ``inverse_transform`` renames back.
        DataTree inputs put each piece in its own node under the source.
        """
        var = self.source.split("/")[-1]
        if isinstance(X, xr.Dataset):
            # Dataset inputs: pieces become sibling variables named
            # ``{source}_{piece}``. A Dataset cannot hold two variables with
            # disjoint values on the same dim (merge outer-joins), so each
            # piece's split dim is renamed ``{dim}_{piece}`` — ``inverse_
            # transform`` renames back and restores the original coordinates.
            out: dict[str, xr.DataArray] = {}
            for key, da in flatten(X).items():
                if key == var:
                    continue  # replaced by the pieces
                out[key] = da
            for piece, sel in self.pieces.items():
                piece_da = X[var].sel({self.split_dim: list(sel[self.split_dim])})
                piece_da = piece_da.rename({self.split_dim: f"{self.split_dim}_{piece}"})
                out[f"{var}_{piece}"] = piece_da.rename(f"{var}_{piece}")
            return reassemble(X, out)
        if not isinstance(X, xr.DataTree):
            raise TypeError("Splitter works on Dataset or DataTree inputs")
        node = _node_of(X, self.source)
        full = node.dataset[var]
        out = {}
        for key, da in flatten(X).items():
            if key == self.source:
                continue  # replaced by the pieces
            out[key] = da
        for piece, sel in self.pieces.items():
            piece_da = full.sel({self.split_dim: list(sel[self.split_dim])})
            out[f"{self.source}/{piece}"] = piece_da.rename(piece)
        return reassemble(X, out)

    def inverse_transform(self, X: xr.DataTree | xr.Dataset, state: xr.Dataset):
        """Recombine the pieces and restore the original coordinate order."""
        dim = self.split_dim
        stored = list(state[f"labels_{dim}"].values)
        if isinstance(X, xr.Dataset):
            var = self.source.split("/")[-1]
            pieces = []
            for piece in state.attrs["pieces"]:
                piece_var = f"{var}_{piece}"
                if piece_var not in X.data_vars:
                    continue
                pieces.append(X[piece_var].rename({f"{dim}_{piece}": dim}))
            combined = xr.concat(pieces, dim=dim).reindex({dim: stored})
            out = {key: da for key, da in flatten(X).items() if not key.startswith(f"{var}_")}
            out[var] = combined.rename(var)
            return reassemble(X, out)
        parent = _node_of(X, self.source)
        pieces_by_order = [
            next(iter(parent.children[piece].dataset.data_vars.values())) for piece in state.attrs["pieces"]
        ]
        combined = xr.concat(pieces_by_order, dim=dim).reindex({dim: stored})
        out = {key: da for key, da in flatten(X).items() if not key.startswith(f"{self.source}/")}
        out[self.source] = combined.rename(self.source)
        return reassemble(X, out)

VariableDropper

Drop whole variables and/or coordinate values.

Dropping is NOT invertible. Two orthogonal modes, usable together:

  • vars: names of whole variables to remove. This is the drop verb for a Dataset — sklego's ColumnDropper over drop_vars.
  • labels: {dim: labels} to remove from that coordinate dataset-wide (drop_sel). Contrast Selector, which subsets within one variable and leaves the coordinate itself intact.

The motivating workflow is pruning raw inputs once a derived metric supersedes them::

Pipeline(
    [
        ("derive", Derive(name="tv_per_control", left=..., right=..., op="ratio")),
        ("prune", Dropper(vars=["media", "controls"])),
    ]
)

fit is stateless but validates, following sklego: unknown variables/labels raise KeyError, dropping every variable raises ValueError (a pipeline that erases its input is a config bug), and non-Dataset input raises TypeError (DataTree deferred).

class VariableDropper:
    """Drop whole variables and/or coordinate values.

    Dropping is NOT invertible. Two orthogonal modes, usable together:

    - ``vars``: names of whole variables to remove. This is the drop verb
      for a Dataset — sklego's ``ColumnDropper`` over ``drop_vars``.
    - ``labels``: ``{dim: labels}`` to remove from that coordinate
      dataset-wide (``drop_sel``). Contrast ``Selector``, which subsets
      *within one variable* and leaves the coordinate itself intact.

    The motivating workflow is pruning raw inputs once a derived metric
    supersedes them::

        Pipeline(
            [
                ("derive", Derive(name="tv_per_control", left=..., right=..., op="ratio")),
                ("prune", Dropper(vars=["media", "controls"])),
            ]
        )

    ``fit`` is stateless but validates, following sklego: unknown
    variables/labels raise ``KeyError``, dropping every variable raises
    ``ValueError`` (a pipeline that erases its input is a config bug), and
    non-``Dataset`` input raises ``TypeError`` (DataTree deferred).
    """

    def __init__(
        self,
        vars: str | list[str] | None = None,
        labels: Mapping[str, str | list[str]] | None = None,
    ):
        """Take the drop lists; at least one of ``vars``/``labels`` is required."""
        self.vars = [vars] if isinstance(vars, str) else (list(vars) if vars else [])
        self.labels = {
            dim: [value] if isinstance(value, str) else list(value) for dim, value in (labels or {}).items()
        }
        if not self.vars and not self.labels:
            raise ValueError("Dropper needs at least one of vars= or labels= to drop")

    def to_spec_kwargs(self) -> dict[str, Any]:
        """Serialize the drop lists into plain-dict kwargs."""
        return {
            "vars": list(self.vars),
            "labels": {dim: list(values) for dim, values in self.labels.items()},
        }

    @classmethod
    def describe(cls, spec):
        """One non-affine dropped row per removed variable."""
        from .serialize import _recipe_line

        recipe = _recipe_line({"kind": "VariableDropper", "kwargs": spec})
        return [
            {
                "variable": variable,
                "role": "dropped",
                "kind": "VariableDropper",
                "recipe": recipe,
                "op": "",
                "affine": False,
            }
            for variable in spec["vars"]
        ]

    def fit(self, X: Any) -> xr.Dataset:
        """Validate the drop lists against ``X``, then record them for audit."""
        if not isinstance(X, xr.Dataset):
            raise TypeError(
                f"Dropper takes an xr.Dataset, got {type(X).__name__}; DataTree node drops are deferred."
            )
        missing = [name for name in self.vars if name not in X.data_vars]
        if missing:
            raise KeyError(f"{missing} variable(s) not in dataset")
        bad_labels = {
            dim: [value for value in values if value not in X.coords[dim].values]
            for dim, values in self.labels.items()
        }
        bad_labels = {dim: values for dim, values in bad_labels.items() if values}
        if bad_labels:
            raise KeyError(f"{bad_labels} label(s) not in dataset coords")
        remaining = set(X.data_vars) - set(self.vars)
        if not remaining:
            raise ValueError(f"Dropping {self.vars} would result in an empty dataset")
        # JSON strings, not dicts: state attrs must survive to_netcdf (the
        # whole drop recipe lives in the spec; this copy is for auditing)
        return _stamp_state(
            as_state(
                "drop",
                None,
                scaled_from=None,
                attrs={
                    "dropped_vars": json.dumps(self.vars),
                    "dropped_labels": json.dumps(self.labels),
                },
            ),
            self,
        )

    def transform(self, X: Any, state: xr.Dataset) -> Any:
        """``drop_vars`` the listed variables, then ``drop_sel`` the labels."""
        out = X
        if self.vars:
            out = out.drop_vars(self.vars)
        if self.labels:
            out = out.drop_sel(**self.labels)
        return out

    def inverse_transform(self, X: Any, state: xr.Dataset) -> Any:
        """Dropping removes data; raising is the contract, not a guess."""
        raise NonInvertibleError(
            "Dropping removes data; inverse_transform cannot restore it. Keep the "
            "raw variables as separate pipeline inputs, or use Splitter for a "
            "reversible split."
        )

@register
class ZScore(_DataDerivedScaler):
    """Standardize along the reduce dims: mean and std (ddof=0).

    Matches sklearn's ``StandardScaler`` (which stores ``mean_``/``scale_``)
    when the kept dim is the feature axis. NaNs are skipped in the
    statistics. The affine roles map center→mean, scale→std.
    """

    kind = "zscore"
    PAIR: ClassVar[dict[str, str]] = {"center": "mean", "scale": "std"}

    def _stats(self, da: Any) -> dict[str, Any]:
        dims = self._reduce_dims(da)
        return {
            "mean": da.mean(dim=dims, skipna=True),
            "std": da.std(dim=dims, ddof=0, skipna=True),
        }

def var_of(node: xr.DataTree) -> str:
    """Sole variable name of a node's dataset (single-var nodes assumed)."""
    vars_ = list(node.dataset.data_vars)
    if len(vars_) != 1:
        raise ValueError(f"Node {node.path!r} must hold exactly one variable; has {vars_}")
    return str(vars_[0])
